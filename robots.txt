/**
 * Ultimate Frontend Security Guard Script
 * Multi-layer Defense in Depth Architecture
 */
(function () {
  'use strict';

  // ==========================================
  // 1. SESSION & AUTHENTICATION GUARD
  // ==========================================
  // ตรวจสอบเซสชันการเข้าใช้งานเบื้องต้น
  (function checkSession() {
    const isVerified = sessionStorage.getItem('isBossVerified');
    if (!isVerified || isVerified !== 'true') {
      // หากไม่มีเซสชัน ให้ล้างหน้าจอและดีดออกทันที
      document.documentElement.innerHTML = '';
      window.location.replace('index.html');
    }
  })();

  // ==========================================
  // 2. KEYBOARD & INPUT HARDENING
  // ==========================================
  // บล็อกการคลิกขวา (Context Menu)
  document.addEventListener('contextmenu', function (e) {
    e.preventDefault();
    return false;
  }, false);

  // บล็อก Shortcut Keys (F12, Inspect Element, View Source, Save)
  document.addEventListener('keydown', function (e) {
    // บล็อก F12
    if (e.key === 'F12' || e.keyCode === 123) {
      e.preventDefault();
      return false;
    }

    // บล็อก Ctrl+Shift+I, Ctrl+Shift+J, Ctrl+Shift+C, Ctrl+Shift+K (DevTools)
    if (e.ctrlKey && e.shiftKey && ['I', 'i', 'J', 'j', 'C', 'c', 'K', 'k'].includes(e.key)) {
      e.preventDefault();
      return false;
    }

    // บล็อก Ctrl+U (View Source) และ Ctrl+S (Save Page)
    if (e.ctrlKey && ['U', 'u', 'S', 's'].includes(e.key)) {
      e.preventDefault();
      return false;
    }

    // บล็อก Cmd (Mac) Shortcut สำหรับระบบ macOS
    if (e.metaKey && e.altKey && ['I', 'i', 'J', 'j', 'C', 'c', 'U', 'u'].includes(e.key)) {
      e.preventDefault();
      return false;
    }
  }, false);

  // บล็อกการลากคุมดำข้อความ และการ Drag & Drop
  document.addEventListener('dragstart', function (e) {
    e.preventDefault();
  }, false);

  document.addEventListener('selectstart', function (e) {
    // อนุญาตให้พิมพ์ข้อความในกล่อง input หรือ textarea ได้ตามปกติ
    if (e.target.tagName !== 'INPUT' && e.target.tagName !== 'TEXTAREA') {
      e.preventDefault();
    }
  }, false);

  // ==========================================
  // 3. CONSOLE & DEBUGGER ANTI-TAMPER
  // ==========================================
  // ปิดการประมวลผล console ทั้งหมด เพื่อห้ามใช้คำสั่งฉีดโค้ด (Script Injection)
  const disableConsole = function () {
    const noop = function () {};
    if (window.console) {
      window.console.log = noop;
      window.console.warn = noop;
      window.console.error = noop;
      window.console.info = noop;
      window.console.debug = noop;
      window.console.table = noop;
      window.console.trace = noop;
    }
  };
  disableConsole();

  // ตรวจจับการเปิด Developer Tools ด้วย Debugger Loop
  setInterval(function () {
    const startTime = performance.now();
    
    // คำสั่งสร้าง Breakpoint ดักแฮกเกอร์ที่เปิด DevTools ทิ้งไว้
    (function () {}['constructor']('debugger')());
    
    const endTime = performance.now();

    // หากดีเลย์เกิน 100ms แสดงว่าหน้าจอติด Breakpoint ของ DevTools
    if (endTime - startTime > 100) {
      sessionStorage.removeItem('isBossVerified');
      document.body.innerHTML = '<div style="color:red;font-size:24px;text-align:center;margin-top:20vh;font-family:sans-serif;"><h1>Unauthorized Debugger Detected</h1><p>Access Revoked.</p></div>';
      setTimeout(function () {
        window.location.replace('index.html');
      }, 1000);
    }
  }, 1000);

  // ==========================================
  // 4. ANTI-CLICKJACKING (IFRAME GUARD)
  // ==========================================
  // ป้องกันการแอบนำหน้าเว็บไปแสดงบน <iframe> ของเว็บอื่น
  if (window.top !== window.self) {
    window.top.location.href = window.self.location.href;
  }

})();
